CREATE DATABASE IF NOT EXISTS testpaper
  DEFAULT CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci;
USE testpaper;

CREATE TABLE questions (
  id            VARCHAR(32) NOT NULL,
  stem          TEXT,
  multi_correct TINYINT(1)  DEFAULT NULL,
  created_at    DATETIME(6) DEFAULT NULL,
  PRIMARY KEY (id)
);

CREATE TABLE options (
  option_id   VARCHAR(35) NOT NULL,
  question_id VARCHAR(32) NOT NULL,
  option_text TEXT,
  is_correct  TINYINT(1)  DEFAULT NULL,
  PRIMARY KEY (option_id),
  KEY fk_options_question (question_id),
  CONSTRAINT fk_options_question FOREIGN KEY (question_id)
    REFERENCES questions (id) ON DELETE CASCADE
);

CREATE TABLE tags (
  id          VARCHAR(255) NOT NULL,
  name        VARCHAR(255) NOT NULL,
  description TEXT,
  created_at  DATETIME(6)  DEFAULT NULL,
  PRIMARY KEY (id),
  UNIQUE KEY uq_tags_name (name)
);

CREATE TABLE tag_questions (
  id          BIGINT       NOT NULL AUTO_INCREMENT,
  tag_id      VARCHAR(255) NOT NULL,
  question_id VARCHAR(32)  NOT NULL,
  created_at  DATETIME(6)  DEFAULT NULL,
  PRIMARY KEY (id),
  UNIQUE KEY uq_tag_question (tag_id, question_id),
  CONSTRAINT fk_tag_questions_tag FOREIGN KEY (tag_id)
    REFERENCES tags (id) ON DELETE CASCADE
);

CREATE TABLE question_papers (
  id         VARCHAR(255) NOT NULL,
  name       VARCHAR(255) NOT NULL,
  created_at DATETIME(6)  DEFAULT NULL,
  PRIMARY KEY (id),
  UNIQUE KEY uq_question_papers_name (name)
);

CREATE TABLE question_paper_questions (
  question_paper_id VARCHAR(255) NOT NULL,
  question_id       VARCHAR(255) DEFAULT NULL,
  KEY fk_qpq_paper (question_paper_id),
  CONSTRAINT fk_qpq_paper FOREIGN KEY (question_paper_id)
    REFERENCES question_papers (id) ON DELETE CASCADE
);

CREATE TABLE users (
  username      VARCHAR(255) NOT NULL,
  name          VARCHAR(255) NOT NULL,
  email         VARCHAR(255) NOT NULL,
  date_of_birth DATE         DEFAULT NULL,
  created_at    DATETIME(6)  DEFAULT NULL,
  updated_at    DATETIME(6)  DEFAULT NULL,
  accesstype    VARCHAR(255) NOT NULL DEFAULT 'Student',
  PRIMARY KEY (username),
  UNIQUE KEY uq_users_email (email)
);

CREATE TABLE test_user_responses (
  user_id           VARCHAR(255) NOT NULL,
  question_paper_id VARCHAR(255) NOT NULL,
  answers_json      TEXT,
  created_at        DATETIME(6)  DEFAULT NULL,
  updated_at        DATETIME(6)  DEFAULT NULL,
  PRIMARY KEY (user_id, question_paper_id)
);

CREATE TABLE people (
  id   BIGINT NOT NULL,
  name VARCHAR(255) DEFAULT NULL,
  PRIMARY KEY (id)
);
